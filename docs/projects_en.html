Newton Iteration Core 
def solver(xi, coeff):
                                maxError = 1e-6
                                error = math.inf
                                while (error > maxError):
                                    df = poly(xi, dpoly(coeff))
                                    if (df != 0):
                                        xn = xi - poly(xi, coeff)/df
                                    else:
                                        xn = xi - poly(xi, coeff)/\
                                            random.uniform(-1e-12, 1e-12)
                                    error = abs(xn - xi)
                                    xi = xn
                                return format(xn, '.4f')

Manual CDF Lookup Construction 
hist_a = np.bincount(image.ravel(), minlength=256)
                                cdf_a  = np.cumsum(hist_a / hist_a.sum())

                                hist_b = np.bincount(reference.ravel(), minlength=256)
                                cdf_b  = np.cumsum(hist_b / hist_b.sum())
